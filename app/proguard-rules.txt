# The FFmpeg audio decoder's native code calls back into its Java classes and Media3's decoder
# buffers by name, which R8 can't see: keep them whole.
-keep class io.github.anilbeesetti.nextlib.** { *; }
-keep class androidx.media3.decoder.** { *; }

# WireGuard's Go library is reached through JNI by class and method name.
-keep class com.wireguard.android.backend.GoBackend { *; }
-keep class com.wireguard.android.backend.GoBackend$* { *; }

# Stack traces in logcat stay readable.
-keepattributes SourceFile,LineNumberTable
-dontobfuscate
